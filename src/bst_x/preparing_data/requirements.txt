# Prepare training data — rtmlib pose extraction + collation into batch arrays.
# Python 3.11+ (validated on 3.13). Run once per dataset build, then switch to
# the BST training venv.
#
# The 2D extraction path (raw_extract.py, prepare_train_on_shuttleset's
# detect_players_2d) migrated off OpenMMLab onto rtmlib: onnxruntime-backed,
# numpy-2 clean, no source build, no separate-venv numpy<2 pin. Same model
# family as before (RTMDet-nano person + RTMPose-L body7 COCO-17). Versions below
# are the set the migration gate ladder (G1-G6) was validated against.
#
# --- Manual install steps (can't go in requirements.txt) ---
# CPU:  pip install -r requirements.txt
# GPU (production re-extract on the CUDA box): swap onnxruntime -> onnxruntime-gpu
#   pip install -r requirements.txt
#   pip uninstall -y onnxruntime && pip install onnxruntime-gpu==1.27.0
#   (rtmlib installs the CPU onnxruntime by default; confirm CUDAExecutionProvider
#    is present after the swap.)
#
# torch is NOT required by the production raw_extract path (Batch 2 dropped it;
# onnxruntime manages its own memory). It is needed ONLY if you run the secondary
# prepare_train_on_shuttleset module path (it imports torch at module load for the
# shared 2D/3D runner) — install any modern CPU/CUDA build separately, e.g.:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#
# The dormant 3D pose path (detect_players_3d / --use-3d-pose) still needs the
# OpenMMLab stack — see requirements-legacy-3d.txt (install in a separate venv;
# its numpy<2 pin cannot coexist with the numpy>=2 here).

# rtmlib pose extraction (RTMDet + RTMPose over onnxruntime)
rtmlib==0.0.15                   # low-level RTMDet/RTMPose classes; pin (repo main drifts from PyPI)
onnxruntime==1.27.0              # CPU inference (deterministic); use onnxruntime-gpu==1.27.0 on the CUDA box

# Standard libraries
numpy==2.4.6                     # numpy-2 clean now that OpenMMLab is off the 2D path
opencv-python==4.13.0.92         # cv2: video decode + the RGB-fix cvtColor in the adapter
pandas==3.0.3                    # collation; G5 verified byte-identical downstream on 3.0.3
scipy==1.17.1                    # court projection / geometry helpers
tqdm==4.68.3                     # progress bars
parse==1.22.1                    # pure Python, no deps
